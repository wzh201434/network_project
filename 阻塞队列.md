##### 根据cv mutex实现最简单、不考虑任何情况的生产者、消费者模型
class BlockDeque {
    // 阻塞队列
    // 生产者、消费者模型
    std::deque<int> data;
    std::mutex mtx;
    std::condition_variable cv;
  public:
    void push(int value);
    optional<int> pop();
    BlockDeque() = default;
    ~BlockDeque() = default;
};
实现：
void BlockDeque::push(int value) {
    std::lock_guard<std::mutex> lock(mtx);
    data.push_back(value);
    cv.notify_one();
}
optional<int> BlockDeque::pop() {
    std::unique_lock<std::mutex> lock(mtx);
    cv.wait(lock, [this]() { return !data.empty(); });
    if (data.empty()) {
        return std::nullopt;
    }
    int value = data.front();
    data.pop_front();
    return value;
}

main中简单调用：
std::vector<std::string> splitStr(const std::string &str, char delim) {
    std::vector<std::string> res;
    size_t start = 0, pos = str.find(delim);
    while (pos != std::string::npos) {
        res.push_back(str.substr(start, pos - start));
        start = pos + 1;
        pos = str.find(delim, start);
    }
    res.push_back(str.substr(start));
    return res;
}
int main() {
    BlockDeque bd;
    std::string line;
    while (getline(std::cin, line)) {
        std::vector<std::string> tokens = splitStr(line, ' ');

        if (tokens[0] == "push") {
            if (tokens.size() < 2) {
                std::cerr << "Error: push command requires a value."
                          << std::endl;
                continue;
            }
            bd.push(stoi(tokens[1]));
        } else if (tokens[0] == "pop") {
            auto value = bd.pop();
            if (value.has_value()) {
                std::cout << value.value() << std::endl;
            } else {
                std::cout << "empty" << std::endl;
            }
        }
    }
}
###### 存在哪些问题
这是一个单生产者、单消费者，生产者不停生产、消费者不停消费的理想实现。
1. 单线程下，一旦生产者停止生产，消费者会一直阻塞在pop()函数中，无法退出。
2. 未设置存储边界，如果生产者一直生产，队列会一直增长直到内存耗尽；
